<script>
async function loadTours() {
  const response = await fetch('https://www.tourninja.io/api/public/tours?apiKey=tourninja-showcase-2-amontour&companyId=2');
  const data = await response.json();
  
  if (!data.success) {
    console.error('Erreur API');
    return;
  }
  
  // UTILISER DIRECTEMENT tour.image - C'EST UNE URL VALIDE !
  const html = data.tours.map(tour => `
    <div class="tour-card">
      <img 
        src="${tour.image}"  <!-- CETTE URL FONCTIONNE ! -->
        alt="${tour.name}"
        onerror="console.error('Erreur image:', '${tour.image}')"
        onload="console.log('Image chargée:', '${tour.name}')"
      />
      <h3>${tour.name}</h3>
      <p>${tour.description}</p>
      <p>Prix: ${tour.price} ${tour.currency}</p>
      <a href="${tour.bookingUrl}">Réserver</a>
    </div>
  `).join('');
  
  document.getElementById('tours-container').innerHTML = html;
}
loadTours();
</script>

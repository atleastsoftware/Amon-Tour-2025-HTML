// Code pour le site vitrine d'Amon Tour
async function loadTours() {
  const response = await fetch('https://votre-domaine-tourninja.com/api/public/company-links/2');
  const data = await response.json();
  
  return data.links.map(tour => ({
    id: tour.token,
    name: tour.tourName,
    price: tour.pricePerPerson,
    currency: tour.currency,
    duration: tour.days + (tour.days > 1 ? ' jours' : ' jour'),
    capacity: tour.maxParticipants + ' personnes max',
    
    // Images de présentation
    mainImage: tour.primaryImage,
    gallery: tour.images,
    hasPhotos: tour.hasImages,
    photoCount: tour.imageCount,
    
    // Liens de réservation
    bookNow: tour.bookingUrl,
    viewDetails: tour.detailsUrl,
    
    // Informations tarifaires
    adultPrice: tour.pricing.adult,
    childPrice: tour.pricing.children
  }));
}

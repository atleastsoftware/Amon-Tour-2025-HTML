# Guide d'intégration API TourNinja pour Amon Tour

## Vue d'ensemble
Cette documentation vous permet d'intégrer facilement l'API publique TourNinja optimisée sur votre site web Amon Tour pour afficher vos tours et excursions.

## Informations d'accès

### URL de l'API
```
https://www.tourninja.io/api/public/tours
```

### Paramètres requis
- `apiKey`: `tourninja-showcase-2-amontour`
- `companyId`: `2`

### URL complète
```
https://www.tourninja.io/api/public/tours?apiKey=tourninja-showcase-2-amontour&companyId=2
```

### Option 2: Intégration avec jQuery

```html
<script src="https://code.jquery.com/jquery-3.6.0.min.js"></script>
<script>
$(document).ready(function() {
    // Configuration API
    const API_CONFIG = {
        url: 'https://www.tourninja.io/api/public/tours',
        apiKey: 'tourninja-showcase-2-amontour',
        companyId: '2'
    };

    // Charger les tours
    function loadTours() {
        const url = `${API_CONFIG.url}?apiKey=${API_CONFIG.apiKey}&companyId=${API_CONFIG.companyId}`;
        
        $.ajax({
            url: url,
            method: 'GET',
            dataType: 'json',
            success: function(data) {
                if (data.success && data.tours) {
                    displayTours(data.tours);
                } else {
                    $('#tours-container').html('<div class="error">Erreur lors du chargement des tours</div>');
                }
            },
            error: function(xhr, status, error) {
                console.error('Erreur API:', error);
                $('#tours-container').html('<div class="error">Erreur de connexion à l\'API</div>');
            }
        });
    }

    // Afficher les tours
    function displayTours(tours) {
        const container = $('#tours-container');
        container.empty();
        
        tours.forEach(function(tour) {
            const tourCard = $(`
                <div class="tour-card">
                    ${tour.image ? `<img src="${tour.image}" alt="${tour.name}" class="tour-image">` : ''}
                    <h3 class="tour-title">${tour.name}</h3>
                    <p class="tour-description">${tour.description}</p>
                    <div class="tour-price">${tour.price} ${tour.currency}</div>
                    <div class="tour-details">
                        <span>Durée: ${tour.duration} jour${tour.duration > 1 ? 's' : ''}</span>
                        <span>Max: ${tour.maxParticipants} pers.</span>
                        <span>Type: ${tour.type}</span>
                    </div>
                    <button class="book-button" data-tour-id="${tour.id}">
                        Réserver maintenant
                    </button>
                </div>
            `);
            container.append(tourCard);
        });

        // Gestionnaire de clic pour les boutons de réservation
        $('.book-button').click(function() {
            const tourId = $(this).data('tour-id');
            const bookingUrl = `https://www.tourninja.io/book/${tourId}`;
            window.open(bookingUrl, '_blank');
        });
    }

    // Charger les tours
    loadTours();
});
</script>
```

## Gestion des erreurs

```javascript
// Exemple de gestion d'erreur robuste
async function loadToursWithErrorHandling() {
    try {
        const response = await fetch(API_URL, {
            method: 'GET',
            headers: {
                'Accept': 'application/json'
            },
            timeout: 10000 // 10 secondes
        });
        
        if (!response.ok) {
            throw new Error(`HTTP ${response.status}: ${response.statusText}`);
        }
        
        const data = await response.json();
        
        if (!data.success) {
            throw new Error('API a retourné une erreur');
        }
        
        return data.tours;
        
    } catch (error) {
        console.error('Erreur chargement tours:', error);
        
        // Afficher un message d'erreur utilisateur
        document.getElementById('tours-container').innerHTML = `
            <div class="error-message">
                <h3>Tours temporairement indisponibles</h3>
                <p>Nous rencontrons des difficultés techniques. Veuillez réessayer dans quelques minutes.</p>
                <button onclick="loadToursWithErrorHandling()">Réessayer</button>
            </div>
        `;
        
        return [];
    }
}
```

## Optimisations recommandées

### 1. Mise en cache
```javascript
// Cache les résultats pendant 1 heure
const CACHE_DURATION = 60 * 60 * 1000; // 1 heure en millisecondes

function getCachedTours() {
    const cached = localStorage.getItem('tourninja_tours');
    if (cached) {
        const data = JSON.parse(cached);
        if (Date.now() - data.timestamp < CACHE_DURATION) {
            return data.tours;
        }
    }
    return null;
}

function setCachedTours(tours) {
    localStorage.setItem('tourninja_tours', JSON.stringify({
        tours: tours,
        timestamp: Date.now()
    }));
}
```

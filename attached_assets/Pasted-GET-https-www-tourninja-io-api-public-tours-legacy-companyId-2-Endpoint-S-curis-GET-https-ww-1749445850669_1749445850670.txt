// Récupération avec cache 6h
const response = await fetch('https://www.tourninja.io/api/public/tours/legacy?companyId=2');
const data = await response.json();
// Groupement par destination
const toursByDestination = data.tours.reduce((acc, tour) => {
  const dest = tour.destination || 'Autres';
  if (!acc[dest]) acc[dest] = [];
  acc[dest].push(tour);
  return acc;
}, {});
// Affichage prix intelligent
const displayPrice = (tour) => {
  if (tour.priceTable) return "À partir de " + Math.min(...tour.priceTable.map(p => p.price)) + " THB";
  if (tour.price > 0) return tour.price + " THB";
  return "Prix sur demande";
};
